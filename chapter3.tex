\section{Database Conceptual Design}
In this section, we construct the conceptual model (ERD) for the 
\textbf{Drone Delivery Management Platform}.
The main entities include:

\begin{itemize}
    \item \textbf{Customer}(\underline{CustomerID}, CustomerName, PhoneNumber, Address)
    \item \textbf{DeliveryOrder}(\underline{OrderID}, CustomerID, Status, ETA)
    \item \textbf{Package}(\underline{PackageID}, OrderID, Weight, Dimensions)
    \item \textbf{Confirmation}(\underline{ConfirmationID}, OrderID, ReceiverName, TimeConfirmed)
    \item \textbf{LandingStation}(\underline{StationID}, Location, Capacity)
    \item \textbf{Tracking}(\underline{TrackingID}, OrderID, CurrentLocation, Timestamp)
    \item \textbf{Drone}(\underline{DroneID}, Status)
    \item \textbf{DeliveryWorkflow}(\underline{WorkflowID}, OrderID, CurrentStatus, AssignedUserID, Timestamp)
    \item \textbf{User}(\underline{UserID}, UserRole, UserPassword, PhoneNumber)
    \item \textbf{AuditLog}(\underline{LogID}, UserID, ActionType, ActionTime, AffectedRecord)
\end{itemize}

\section{Relationships}
\begin{itemize}
    \item Customer $\to$ DeliveryOrder (1:N)
    \item DeliveryOrder $\to$ Package (1:N)
    \item DeliveryOrder $\to$ Confirmation (1:1)
    \item DeliveryOrder $\leftrightarrow$ LandingStation (N:N)
    \item DeliveryOrder $\to$ Tracking (1:N)
    \item DeliveryOrder $\to$ Drone (1:1)
    \item DeliveryOrder $\to$ DeliveryWorkflow (1:N)
    \item DeliveryWorkflow $\to$ User (N:1)
    \item User $\to$ AuditLog (1:N)
\end{itemize}

\section{Logical Design}
From the conceptual model, we derive the relational schema:

\begin{itemize}
    \item Customer(CustomerID PK, CustomerName, PhoneNumber, Address)
    \item DeliveryOrder(OrderID PK, CustomerID FK, Status, ETA)
    \item Package(PackageID PK, OrderID FK, Weight, Dimensions)
    \item Confirmation(ConfirmationID PK, OrderID FK, ReceiverName, TimeConfirmed)
    \item LandingStation(StationID PK, Location, Capacity)
    \item Tracking(TrackingID PK, OrderID FK, CurrentLocation, Timestamp)
    \item Drone(DroneID PK, Status)
    \item DeliveryWorkflow(WorkflowID PK, OrderID FK, CurrentStatus, AssignedUserID FK, Timestamp)
    \item User(UserID PK, UserRole, UserPassword, PhoneNumber)
    \item AuditLog(LogID PK, UserID FK, ActionType, ActionTime, AffectedRecord)
\end{itemize}

\section{Physical Design}
The tables are implemented in SQL as follows:

\begin{verbatim}
CREATE TABLE Customer (
    CustomerID SERIAL PRIMARY KEY,
    CustomerName VARCHAR(100),
    PhoneNumber VARCHAR(15) UNIQUE,
    Address TEXT
);

CREATE TABLE DeliveryOrder (
    OrderID SERIAL PRIMARY KEY,
    CustomerID INT REFERENCES Customer(CustomerID),
    Status VARCHAR(20),
    ETA TIMESTAMP
);

CREATE TABLE Package (
    PackageID SERIAL PRIMARY KEY,
    OrderID INT REFERENCES DeliveryOrder(OrderID),
    Weight DECIMAL(10,2) CHECK (Weight > 0),
    Dimensions VARCHAR(50)
);

CREATE TABLE Confirmation (
    ConfirmationID SERIAL PRIMARY KEY,
    OrderID INT REFERENCES DeliveryOrder(OrderID),
    ReceiverName VARCHAR(100),
    TimeConfirmed TIMESTAMP
);

CREATE TABLE LandingStation (
    StationID SERIAL PRIMARY KEY,
    Location VARCHAR(100),
    Capacity INT CHECK (Capacity >= 0)
);

CREATE TABLE Tracking (
    TrackingID SERIAL PRIMARY KEY,
    OrderID INT REFERENCES DeliveryOrder(OrderID),
    CurrentLocation VARCHAR(100),
    Timestamp TIMESTAMP
);

CREATE TABLE Drone (
    DroneID SERIAL PRIMARY KEY,
    Status VARCHAR(20) CHECK (Status IN ('Available','Busy','Maintenance'))
);

CREATE TABLE User (
    UserID SERIAL PRIMARY KEY,
    UserRole VARCHAR(50) CHECK (UserRole IN ('Admin','Operator','CustomerService')),
    UserPassword VARCHAR(255),
    PhoneNumber VARCHAR(15)
);

CREATE TABLE DeliveryWorkflow (
    WorkflowID SERIAL PRIMARY KEY,
    OrderID INT REFERENCES DeliveryOrder(OrderID),
    CurrentStatus VARCHAR(50),
    AssignedUserID INT REFERENCES User(UserID),
    Timestamp TIMESTAMP
);

CREATE TABLE AuditLog (
    LogID SERIAL PRIMARY KEY,
    UserID INT REFERENCES User(UserID),
    ActionType VARCHAR(50),
    ActionTime TIMESTAMP,
    AffectedRecord VARCHAR(100)
);
\end{verbatim}

\section{Normalization and Constraints}
\begin{itemize}
    \item All tables satisfy 1NF, 2NF, and 3NF.
    \item Integrity constraints:
    \begin{itemize}
        \item UNIQUE: customer phone number.
        \item CHECK: order status, drone status, user role.
        \item FOREIGN KEY: relationships between tables as defined in the ERD.
    \end{itemize}
    \item Indexes for query optimization:
    \begin{verbatim}
    CREATE INDEX idx_order_customer ON DeliveryOrder(CustomerID);
    CREATE INDEX idx_tracking_order ON Tracking(OrderID);
    CREATE INDEX idx_workflow_order ON DeliveryWorkflow(OrderID);
    CREATE INDEX idx_audit_user ON AuditLog(UserID);
    \end{verbatim}
\end{itemize}
