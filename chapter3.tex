\section{Thiết kế cơ sở dữ liệu mức khái niệm}\label{s:conceptual}

Phần này trình bày mô hình cơ sở dữ liệu mức khái niệm cho nền tảng SmartDroneDelivery. Mô hình xác định các thực thể chính, thuộc tính của chúng, mối quan hệ giữa các thực thể và bản số của những mối quan hệ đó. Đây là cơ sở để xây dựng lược đồ quan hệ mức logic trong Phần~\ref{s:logical}.

\subsection{Xác định thực thể}\label{s:entities}

Dựa trên các yêu cầu và quy tắc nghiệp vụ được trình bày tại Chương~\ref{c:analysis}, hai mươi ba thực thể sau được xác định. Các thực thể được tổ chức thành bảy nhóm chức năng nhằm bảo đảm bao phủ đầy đủ nhưng không tạo ra dư thừa trong vận hành.

\paragraph{1. Nhóm người dùng và phân quyền.}
\begin{itemize}
    \item \textbf{system\_user} — Lưu thông tin xác thực và trạng thái tài khoản của mọi tác nhân; tên bảng tránh sử dụng từ khóa dành riêng của SQL.
    \item \textbf{role} — Xác định các vai trò vận hành tiêu chuẩn (\texttt{CUSTOMER}, \texttt{DISPATCHER}, \texttt{STATION\_OPERATOR}, \texttt{LOGISTICS\_MANAGER}, \texttt{SYSTEM\_ADMIN}).
    \item \textbf{permission} — Xác định các quyền hạn chi tiết, chẳng hạn \texttt{CREATE\_ORDER}, \texttt{ASSIGN\_DRONE}, \texttt{CONFIRM\_PICKUP}.
    \item \textbf{user\_role} — Bảng liên kết triển khai mối quan hệ nhiều--nhiều giữa người dùng và vai trò.
    \item \textbf{role\_permission} — Bảng liên kết triển khai mối quan hệ nhiều--nhiều giữa vai trò và quyền hạn.
\end{itemize}

\paragraph{2. Nhóm khách hàng và địa chỉ.}
\begin{itemize}
    \item \textbf{customer} — Lưu thông tin hồ sơ khách hàng và có thể liên kết với một tài khoản người dùng hệ thống.
    \item \textbf{customer\_address} — Lưu một hoặc nhiều địa chỉ giao hàng của mỗi khách hàng, bao gồm tọa độ địa lý chính xác phục vụ định vị cho máy bay không người lái.
\end{itemize}

\paragraph{3. Nhóm đơn giao hàng và kiện hàng.}
\begin{itemize}
    \item \textbf{delivery\_order} — Đại diện cho yêu cầu giao hàng của khách hàng, lưu thời gian khách hàng mong muốn, thời gian đến dự kiến và trạng thái vòng đời.
    \item \textbf{package} — Lưu thông số vật lý của kiện hàng (khối lượng và kích thước tính bằng centimét) thuộc một đơn hàng.
    \item \textbf{order\_status\_history} — Bản ghi chỉ cho phép ghi nối tiếp, lưu mọi lần chuyển trạng thái của đơn giao hàng.
\end{itemize}

\paragraph{4. Nhóm quản lý trạm hạ cánh và máy bay.}
\begin{itemize}
    \item \textbf{landing\_station} — Lưu các trạm cất/hạ cánh vật lý, tọa độ và sức chứa máy bay.
    \item \textbf{station\_status\_history} — Lưu lịch sử chuyển trạng thái trạm, chẳng hạn trạng thái bảo trì hoặc sẵn sàng.
    \item \textbf{station\_operator\_assignment} — Lưu việc phân công nhân viên vận hành vào trạm trong những khoảng thời gian xác định.
    \item \textbf{drone} — Lưu số sê-ri, mẫu, trạng thái vận hành, mức pin và tải trọng của máy bay không người lái.
\end{itemize}

\paragraph{5. Nhóm thực hiện giao hàng và dữ liệu đo từ xa.}
\begin{itemize}
    \item \textbf{delivery\_activity} — Thực thể vận hành trung tâm đại diện cho từng lần thực hiện một đơn hàng, liên kết máy bay, trạm khởi hành và trạm đích.
    \item \textbf{package\_station\_event} — Lưu các sự kiện xử lý vật lý của kiện hàng tại trạm hạ cánh như tiếp nhận, xác nhận lấy hàng, chất hàng và dỡ hàng.
    \item \textbf{tracking\_record} — Lưu dữ liệu GPS và dữ liệu đo từ xa tần suất cao được thu thập trong chuyến bay.
    \item \textbf{delivery\_exception} — Lưu các thất bại, bất thường hoặc gián đoạn của chuyến bay trong một hoạt động giao hàng.
\end{itemize}

\paragraph{6. Nhóm xác nhận giao hàng.}
\begin{itemize}
    \item \textbf{delivery\_confirmation} — Lưu bằng chứng bàn giao chính thức sau khi giao hàng thành công, liên kết 1:0..1 với đơn hàng và tham chiếu tới bằng chứng số trong kho lưu trữ đối tượng.
\end{itemize}

\paragraph{7. Nhóm kiểm toán, tuyến đường và hỗ trợ trí tuệ nhân tạo.}
\begin{itemize}
    \item \textbf{audit\_log} — Dấu vết kiểm toán hệ thống không thể sửa đổi, lưu thao tác người dùng, ảnh chụp JSON trước/sau và địa chỉ IP.
    \item \textbf{delivery\_route} — Lưu siêu dữ liệu về đường bay dự kiến của một hoạt động giao hàng.
    \item \textbf{route\_point} — Lưu các điểm hành trình có thứ tự trên một tuyến giao hàng dự kiến.
    \item \textbf{ai\_recommendation} — Lưu bền vững đầu ra của các mô hình đề xuất bằng trí tuệ nhân tạo, gồm đề xuất máy bay và dự đoán ETA.
\end{itemize}

\subsection{Thuộc tính của thực thể}\label{s:attributes}

Phần này xác định các thuộc tính và kiểu dữ liệu của từng thực thể.

\paragraph{system\_user.}
Lưu thông tin xác thực tài khoản và trạng thái xác thực người dùng.
\begin{itemize}
    \item \texttt{user\_id} — Khóa chính (\texttt{BIGINT GENERATED ALWAYS AS IDENTITY}).
    \item \texttt{username} — Tên đăng nhập duy nhất (\texttt{VARCHAR(100)}).
    \item \texttt{password\_hash} — Giá trị băm mật mã của mật khẩu (\texttt{VARCHAR(255)}).
    \item \texttt{email} — Địa chỉ thư điện tử duy nhất (\texttt{VARCHAR(255)}).
    \item \texttt{phone\_number} — Số điện thoại duy nhất (\texttt{VARCHAR(20)}).
    \item \texttt{is\_active} — Cờ cho biết tài khoản đang hoạt động (\texttt{BOOLEAN}).
    \item \texttt{created\_at}, \texttt{updated\_at} — Các mốc thời gian kiểm toán tài khoản (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{role.}
Xác định các vai trò trong hệ thống.
\begin{itemize}
    \item \texttt{role\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{role\_name} — Mã vai trò duy nhất (\texttt{VARCHAR(50)}).
    \item \texttt{description} — Mô tả trách nhiệm của vai trò (\texttt{VARCHAR(255)}).
\end{itemize}

\paragraph{permission.}
Xác định các quyền chức năng.
\begin{itemize}
    \item \texttt{permission\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{permission\_name} — Mã quyền hạn duy nhất (\texttt{VARCHAR(100)}).
    \item \texttt{description} — Nội dung giải thích quyền hạn (\texttt{VARCHAR(255)}).
\end{itemize}

\paragraph{user\_role.}
Liên kết người dùng với các vai trò.
\begin{itemize}
    \item \texttt{user\_id}, \texttt{role\_id} — Khóa chính ghép và đồng thời là các khóa ngoại.
    \item \texttt{assigned\_at} — Thời điểm gán vai trò (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{role\_permission.}
Liên kết vai trò với các quyền hạn.
\begin{itemize}
    \item \texttt{role\_id}, \texttt{permission\_id} — Khóa chính ghép và đồng thời là các khóa ngoại.
\end{itemize}

\paragraph{customer.}
Lưu hồ sơ khách hàng.
\begin{itemize}
    \item \texttt{customer\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{user\_id} — Khóa ngoại tham chiếu \texttt{system\_user} (\texttt{BIGINT UNIQUE NULL}).
    \item \texttt{customer\_name} — Họ và tên (\texttt{VARCHAR(100)}).
    \item \texttt{phone\_number} — Số điện thoại liên hệ chính (\texttt{VARCHAR(20) UNIQUE}).
    \item \texttt{email} — Thư điện tử liên hệ (\texttt{VARCHAR(255) UNIQUE}).
    \item \texttt{created\_at} — Thời điểm đăng ký (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{customer\_address.}
Lưu các địa điểm giao hàng của khách hàng.
\begin{itemize}
    \item \texttt{address\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{customer\_id} — Khóa ngoại tham chiếu \texttt{customer} (\texttt{BIGINT NOT NULL}).
    \item \texttt{address\_label} — Nhãn như ``Nhà'' hoặc ``Văn phòng'' (\texttt{VARCHAR(50)}).
    \item \texttt{address\_line} — Địa chỉ đường phố (\texttt{VARCHAR(255) NOT NULL}).
    \item \texttt{ward}, \texttt{district}, \texttt{city} — Các đơn vị hành chính (\texttt{VARCHAR(100)}).
    \item \texttt{latitude}, \texttt{longitude} — Tọa độ (\texttt{NUMERIC(9,6)}).
    \item \texttt{is\_default} — Cờ địa chỉ mặc định (\texttt{BOOLEAN NOT NULL DEFAULT FALSE}).
    \item \texttt{created\_at} — Thời điểm tạo (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{delivery\_order.}
Đại diện cho các yêu cầu giao hàng.
\begin{itemize}
    \item \texttt{order\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{customer\_id} — Khóa ngoại tham chiếu \texttt{customer} (\texttt{BIGINT NOT NULL}).
    \item \texttt{delivery\_address\_id} — Khóa ngoại tham chiếu \texttt{customer\_address} (\texttt{BIGINT NOT NULL}).
    \item \texttt{order\_status} — Trạng thái vòng đời hiện tại (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{requested\_delivery\_time} — Thời gian giao hàng khách hàng mong muốn (\texttt{TIMESTAMPTZ}).
    \item \texttt{estimated\_arrival\_at} — Thời gian đến do hệ thống dự kiến (\texttt{TIMESTAMPTZ}).
    \item \texttt{created\_at} — Thời điểm tạo đơn (\texttt{TIMESTAMPTZ NOT NULL}).
    \item \texttt{completed\_at}, \texttt{cancelled\_at} — Các thời điểm kết thúc (\texttt{TIMESTAMPTZ}).
    \item \texttt{cancellation\_reason} — Nội dung giải thích nếu đơn bị hủy (\texttt{VARCHAR(255)}).
\end{itemize}

\paragraph{package.}
Lưu thông số kiện hàng.
\begin{itemize}
    \item \texttt{package\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{order\_id} — Khóa ngoại tham chiếu \texttt{delivery\_order} (\texttt{BIGINT NOT NULL}).
    \item \texttt{description} — Mô tả nội dung kiện hàng (\texttt{VARCHAR(500)}).
    \item \texttt{weight\_kg} — Khối lượng tính bằng kilôgam (\texttt{NUMERIC(10,2) > 0}).
    \item \texttt{length\_cm}, \texttt{width\_cm}, \texttt{height\_cm} — Kích thước tính bằng centimét (\texttt{NUMERIC(10,2) > 0}).
    \item \texttt{package\_status} — Mã trạng thái (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{created\_at} — Thời điểm tạo (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{order\_status\_history.}
Duy trì lịch sử vòng đời đơn hàng không thể sửa đổi.
\begin{itemize}
    \item \texttt{status\_history\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{order\_id} — Khóa ngoại tham chiếu \texttt{delivery\_order} (\texttt{BIGINT NOT NULL}).
    \item \texttt{status} — Giá trị trạng thái được ghi nhận (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{changed\_by} — Khóa ngoại tham chiếu \texttt{system\_user} (\texttt{BIGINT}).
    \item \texttt{reason} — Lý do chuyển trạng thái (\texttt{VARCHAR(255)}).
    \item \texttt{changed\_at} — Thời điểm chuyển trạng thái (\texttt{TIMESTAMPTZ NOT NULL}).
\end{itemize}

\paragraph{landing\_station.}
Lưu các trạm hạ cánh vật lý.
\begin{itemize}
    \item \texttt{station\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{station\_code} — Mã trạm duy nhất (\texttt{VARCHAR(50) UNIQUE NOT NULL}).
    \item \texttt{station\_name} — Tên trạm dễ đọc (\texttt{VARCHAR(100) NOT NULL}).
    \item \texttt{address} — Địa chỉ vật lý (\texttt{VARCHAR(255)}).
    \item \texttt{latitude}, \texttt{longitude} — Tọa độ (\texttt{NUMERIC(9,6)}).
    \item \texttt{capacity} — Sức chứa máy bay tối đa (\texttt{INTEGER >= 0}).
    \item \texttt{status} — Trạng thái vận hành (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{created\_at} — Thời điểm tạo (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{station\_status\_history.}
Duy trì lịch sử vận hành của trạm.
\begin{itemize}
    \item \texttt{history\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{station\_id} — Khóa ngoại tham chiếu \texttt{landing\_station} (\texttt{BIGINT NOT NULL}).
    \item \texttt{status} — Trạng thái trạm được ghi nhận (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{changed\_by} — Khóa ngoại tham chiếu \texttt{system\_user} (\texttt{BIGINT}).
    \item \texttt{reason} — Lý do chuyển trạng thái (\texttt{VARCHAR(255)}).
    \item \texttt{changed\_at} — Thời điểm chuyển trạng thái (\texttt{TIMESTAMPTZ NOT NULL}).
\end{itemize}

\paragraph{station\_operator\_assignment.}
Lưu việc bố trí nhân viên vận hành tại các trạm.
\begin{itemize}
    \item \texttt{assignment\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{station\_id} — Khóa ngoại tham chiếu \texttt{landing\_station} (\texttt{BIGINT NOT NULL}).
    \item \texttt{user\_id} — Khóa ngoại tham chiếu \texttt{system\_user}, đại diện cho nhân viên vận hành (\texttt{BIGINT NOT NULL}).
    \item \texttt{assigned\_from}, \texttt{assigned\_to} — Khoảng thời gian phân công có hiệu lực (\texttt{TIMESTAMPTZ}).
    \item \texttt{is\_active} — Cờ cho biết phân công đang có hiệu lực (\texttt{BOOLEAN NOT NULL DEFAULT TRUE}).
    \item \texttt{created\_at} — Thời điểm tạo phân công (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{drone.}
Lưu thông số và trạng thái của máy bay không người lái.
\begin{itemize}
    \item \texttt{drone\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{serial\_number} — Mã định danh vật lý (\texttt{VARCHAR(100) UNIQUE NOT NULL}).
    \item \texttt{model} — Tên mẫu máy bay (\texttt{VARCHAR(100)}).
    \item \texttt{status} — Trạng thái vận hành (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{max\_payload\_kg} — Tải trọng tối đa (\texttt{NUMERIC(10,2) > 0}).
    \item \texttt{battery\_level} — Mức pin hiện tại (\texttt{NUMERIC(5,2)}, từ 0 đến 100).
    \item \texttt{station\_id} — Khóa ngoại tham chiếu trạm \texttt{landing\_station} hiện tại (\texttt{BIGINT NULL}).
    \item \texttt{created\_at} — Thời điểm đăng ký (\texttt{TIMESTAMPTZ}).
\end{itemize}
Lưu ý: Máy bay không lưu trực tiếp \texttt{order\_id}; việc phân công máy bay được thực hiện thông qua \texttt{delivery\_activity}.

\paragraph{delivery\_activity.}
Thực thể trung gian vận hành đại diện cho các lần thực hiện giao hàng.
\begin{itemize}
    \item \texttt{activity\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{order\_id} — Khóa ngoại tham chiếu \texttt{delivery\_order} (\texttt{BIGINT NOT NULL}).
    \item \texttt{drone\_id} — Khóa ngoại tham chiếu \texttt{drone} (\texttt{BIGINT NOT NULL}).
    \item \texttt{pickup\_station\_id} — Khóa ngoại tham chiếu \texttt{landing\_station} khởi hành (\texttt{BIGINT}).
    \item \texttt{destination\_station\_id} — Khóa ngoại tham chiếu \texttt{landing\_station} đích (\texttt{BIGINT}).
    \item \texttt{assigned\_by} — Khóa ngoại tham chiếu \texttt{system\_user} thực hiện phân công (\texttt{BIGINT}).
    \item \texttt{scheduled\_at} — Thời gian dự kiến thực hiện (\texttt{TIMESTAMPTZ}).
    \item \texttt{activity\_status} — Trạng thái thực hiện (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{started\_at}, \texttt{ended\_at} — Các mốc thời gian thực hiện (\texttt{TIMESTAMPTZ}).
    \item \texttt{previous\_activity\_id} — Khóa ngoại tự tham chiếu, liên kết với lần thực hiện thất bại trước đó (\texttt{BIGINT NULL}).
    \item \texttt{created\_at} — Thời điểm tạo (\texttt{TIMESTAMPTZ}).
\end{itemize}

\paragraph{package\_station\_event.}
Lưu quá trình xử lý kiện hàng tại các trạm.
\begin{itemize}
    \item \texttt{event\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{package\_id} — Khóa ngoại tham chiếu \texttt{package} (\texttt{BIGINT NOT NULL}).
    \item \texttt{station\_id} — Khóa ngoại tham chiếu \texttt{landing\_station} (\texttt{BIGINT NOT NULL}).
    \item \texttt{event\_type} — Loại sự kiện xử lý (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{performed\_by} — Khóa ngoại tham chiếu \texttt{system\_user} là nhân viên vận hành trạm (\texttt{BIGINT}).
    \item \texttt{occurred\_at} — Thời điểm xảy ra sự kiện (\texttt{TIMESTAMPTZ NOT NULL}).
    \item \texttt{note} — Ghi chú vận hành (\texttt{VARCHAR(255)}).
\end{itemize}

\paragraph{tracking\_record.}
Lưu dữ liệu đo từ xa tần suất cao trong chuyến bay.
\begin{itemize}
    \item \texttt{tracking\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{activity\_id} — Khóa ngoại tham chiếu \texttt{delivery\_activity} (\texttt{BIGINT NOT NULL}).
    \item \texttt{latitude}, \texttt{longitude} — Tọa độ (\texttt{NUMERIC(9,6)}).
    \item \texttt{altitude\_m} — Độ cao tính bằng mét (\texttt{NUMERIC(10,2) >= 0}).
    \item \texttt{speed\_kmh} — Tốc độ so với mặt đất tính bằng km/h (\texttt{NUMERIC(10,2) >= 0}).
    \item \texttt{battery\_level} — Phần trăm pin (\texttt{NUMERIC(5,2)}).
    \item \texttt{recorded\_at} — Thời điểm thu thập dữ liệu đo từ xa (\texttt{TIMESTAMPTZ NOT NULL}).
\end{itemize}

\paragraph{delivery\_confirmation.}
Bằng chứng giao hàng chính thức.
\begin{itemize}
    \item \texttt{confirmation\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{order\_id} — Khóa ngoại tham chiếu \texttt{delivery\_order} (\texttt{BIGINT UNIQUE NOT NULL}).
    \item \texttt{activity\_id} — Khóa ngoại tham chiếu \texttt{delivery\_activity} thành công (\texttt{BIGINT UNIQUE NOT NULL}).
    \item \texttt{confirmed\_by} — Khóa ngoại tham chiếu \texttt{system\_user} thực hiện xác minh (\texttt{BIGINT}).
    \item \texttt{receiver\_name} — Tên người nhận kiện hàng (\texttt{VARCHAR(100) NOT NULL}).
    \item \texttt{receiver\_phone} — Số điện thoại liên hệ của người nhận (\texttt{VARCHAR(20)}).
    \item \texttt{confirmation\_method} — Mã phương thức xác nhận (\texttt{VARCHAR(30) NOT NULL}).
    \item \texttt{evidence\_object\_key} — Khóa lưu trữ đối tượng của bằng chứng chữ ký số/hình ảnh (\texttt{VARCHAR(255)}).
    \item \texttt{confirmed\_at} — Thời điểm bàn giao (\texttt{TIMESTAMPTZ NOT NULL}).
    \item \texttt{note} — Ghi chú giao hàng (\texttt{VARCHAR(500)}).
\end{itemize}

\paragraph{delivery\_exception.}
Lưu các thất bại và sự cố chuyến bay.
\begin{itemize}
    \item \texttt{exception\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{activity\_id} — Khóa ngoại tham chiếu \texttt{delivery\_activity} (\texttt{BIGINT NOT NULL}).
    \item \texttt{exception\_type} — Nhóm sự cố (\texttt{VARCHAR(50) NOT NULL}).
    \item \texttt{description} — Mô tả chi tiết sự cố (\texttt{TEXT}).
    \item \texttt{detected\_at} — Thời điểm phát hiện sự cố (\texttt{TIMESTAMPTZ NOT NULL}).
    \item \texttt{resolved\_at} — Thời điểm xử lý xong (\texttt{TIMESTAMPTZ}).
    \item \texttt{resolved\_by} — Khóa ngoại tham chiếu \texttt{system\_user} thực hiện xử lý (\texttt{BIGINT}).
    \item \texttt{resolution} — Mô tả biện pháp khắc phục (\texttt{TEXT}).
\end{itemize}

\paragraph{audit\_log.}
Nhật ký kiểm toán bảo mật và thay đổi trên toàn hệ thống.
\begin{itemize}
    \item \texttt{log\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{user\_id} — Khóa ngoại tham chiếu tác nhân \texttt{system\_user} (\texttt{BIGINT NULL}).
    \item \texttt{entity\_name} — Tên bảng đích (\texttt{VARCHAR(100) NOT NULL}).
    \item \texttt{record\_id} — Khóa chính của bản ghi đích (\texttt{BIGINT}).
    \item \texttt{action\_type} — Loại thao tác (\texttt{VARCHAR(50) NOT NULL}).
    \item \texttt{old\_value}, \texttt{new\_value} — Ảnh chụp JSON trước/sau (\texttt{JSONB}).
    \item \texttt{action\_time} — Thời điểm kiểm toán (\texttt{TIMESTAMPTZ NOT NULL}).
    \item \texttt{ip\_address} — Địa chỉ mạng của máy khách (\texttt{INET}).
\end{itemize}

\paragraph{delivery\_route và route\_point.}
Các quỹ đạo bay và điểm hành trình dự kiến.
\begin{itemize}
    \item \texttt{delivery\_route}: \texttt{route\_id} (PK), \texttt{activity\_id} (FK UNIQUE), \texttt{total\_distance\_km}, \texttt{estimated\_duration\_min}, \texttt{created\_at}.
    \item \texttt{route\_point}: \texttt{route\_point\_id} (PK), \texttt{route\_id} (FK), \texttt{sequence\_number}, \texttt{latitude}, \texttt{longitude}, \texttt{altitude\_m}.
\end{itemize}

\paragraph{ai\_recommendation.}
Đầu ra được lưu bền vững từ các dịch vụ phân tích bằng trí tuệ nhân tạo.
\begin{itemize}
    \item \texttt{recommendation\_id} — Khóa chính (\texttt{BIGINT}).
    \item \texttt{order\_id} — Khóa ngoại tham chiếu \texttt{delivery\_order} (\texttt{BIGINT NOT NULL}).
    \item \texttt{activity\_id} — Khóa ngoại tham chiếu \texttt{delivery\_activity} (\texttt{BIGINT NULL}).
    \item \texttt{recommendation\_type} — Nhóm đề xuất (\texttt{VARCHAR(50) NOT NULL}).
    \item \texttt{recommended\_drone\_id} — Khóa ngoại tham chiếu \texttt{drone} được đề xuất (\texttt{BIGINT NULL}).
    \item \texttt{estimated\_duration\_min} — Số phút bay do mô hình dự kiến (\texttt{NUMERIC(10,2)}).
    \item \texttt{confidence\_score} — Mức độ tin cậy (\texttt{NUMERIC(5,4)}, từ 0 đến 1).
    \item \texttt{model\_version} — Mã phiên bản của dịch vụ trí tuệ nhân tạo (\texttt{VARCHAR(50)}).
    \item \texttt{created\_at} — Thời điểm tạo đề xuất (\texttt{TIMESTAMPTZ}).
\end{itemize}

\subsection{Mối quan hệ và bản số}\label{s:relationships}

Mối quan hệ và bản số giữa hai mươi ba thực thể được xác định như sau, tuân thủ nghiêm ngặt các quy tắc nghiệp vụ:

\begin{itemize}
    \item \textbf{customer} 1 : N \textbf{customer\_address} — Một khách hàng sở hữu nhiều địa chỉ; mỗi địa chỉ thuộc về chính xác một khách hàng (BR-01).
    \item \textbf{customer} 1 : N \textbf{delivery\_order} — Một khách hàng tạo nhiều đơn giao hàng; mỗi đơn thuộc về một khách hàng (BR-02).
    \item \textbf{customer\_address} 1 : N \textbf{delivery\_order} — Một địa chỉ có thể là điểm đến của nhiều đơn; mỗi đơn chỉ định một địa chỉ đích (BR-03, BR-04).
    \item \textbf{delivery\_order} 1 : N \textbf{package} — Một đơn chứa một hoặc nhiều kiện hàng; mỗi kiện hàng thuộc về một đơn (BR-06, BR-07).
    \item \textbf{delivery\_order} 1 : N \textbf{order\_status\_history} — Một đơn tích lũy nhiều bản ghi lịch sử trạng thái (BR-10, BR-11).
    \item \textbf{delivery\_order} 1 : N \textbf{delivery\_activity} — Một đơn có thể có nhiều lần thực hiện, gồm lần đầu và các lần được lập lịch lại (BR-19, BR-20).
    \item \textbf{delivery\_order} 1 : 0..1 \textbf{delivery\_confirmation} — Một đơn có tối đa một xác nhận giao hàng thành công (BR-33).
    \item \textbf{delivery\_activity} 1 : 0..1 \textbf{delivery\_confirmation} — Một xác nhận gắn với chính xác một lần thực hiện giao hàng thành công (BR-35).
    \item \textbf{delivery\_activity} 0..1 : 1 \textbf{delivery\_activity} — Hoạt động được lập lịch lại liên kết với lần thực hiện trước đó thông qua \texttt{previous\_activity\_id} (BR-22).
    \item \textbf{drone} 1 : N \textbf{delivery\_activity} — Một máy bay phục vụ nhiều hoạt động giao hàng theo thời gian, nhưng tại mỗi thời điểm chỉ có một chuyến bay đang hoạt động (BR-15, BR-16).
    \item \textbf{landing\_station} 1 : N \textbf{drone} — Một trạm có thể chứa nhiều máy bay đang đỗ (BR-25).
    \item \textbf{landing\_station} 1 : N \textbf{station\_status\_history} — Một trạm bảo toàn các thay đổi trạng thái vận hành (BR-26).
    \item \textbf{landing\_station} 1 : N \textbf{station\_operator\_assignment} — Một trạm có nhiều lượt phân công nhân viên theo thời gian (BR-27).
    \item \textbf{system\_user} 1 : N \textbf{station\_operator\_assignment} — Một người dùng là nhân viên vận hành có thể được phân công tới nhiều trạm theo thời gian (BR-27).
    \item \textbf{package} 1 : N \textbf{package\_station\_event} — Một kiện hàng trải qua nhiều sự kiện xử lý tại trạm như tiếp nhận, lấy hàng và chất hàng (BR-29, BR-30).
    \item \textbf{landing\_station} 1 : N \textbf{package\_station\_event} — Một trạm xử lý sự kiện của nhiều kiện hàng (BR-31).
    \item \textbf{delivery\_activity} 1 : N \textbf{tracking\_record} — Mỗi hoạt động tạo ra một chuỗi điểm dữ liệu đo từ xa không thể sửa đổi (BR-12).
    \item \textbf{delivery\_activity} 1 : N \textbf{delivery\_exception} — Một hoạt động có thể gặp không, một hoặc nhiều sự cố ngoại lệ (BR-13, BR-21).
    \item \textbf{delivery\_activity} 1 : 0..1 \textbf{delivery\_route} — Một hoạt động có tối đa một tuyến đường dự kiến.
    \item \textbf{delivery\_route} 1 : N \textbf{route\_point} — Một tuyến đường gồm một chuỗi điểm hành trình có thứ tự.
    \item \textbf{delivery\_order} 1 : N \textbf{ai\_recommendation} — Theo thời gian, một đơn có thể nhận đề xuất từ nhiều mô hình trí tuệ nhân tạo.
    \item \textbf{system\_user} N : N \textbf{role} — Được liên kết thông qua \textbf{user\_role}.
    \item \textbf{role} N : N \textbf{permission} — Được liên kết thông qua \textbf{role\_permission}.
    \item \textbf{system\_user} 1 : N \textbf{audit\_log} — Người dùng tạo các bản ghi nhật ký kiểm toán trong toàn hệ thống (BR-37, BR-38).
\end{itemize}

\subsection{Khóa chính, khóa ngoại và khóa ứng viên}

Mỗi bảng cơ sở có một khóa chính thay thế gồm một cột (\texttt{BIGINT GENERATED ALWAYS AS IDENTITY}). Các bảng liên kết (\texttt{user\_role}, \texttt{role\_permission}) sử dụng khóa chính ghép. Các khóa tự nhiên ứng viên được bảo vệ bằng ràng buộc \texttt{UNIQUE}:
\begin{itemize}
    \item \texttt{system\_user.username}, \texttt{system\_user.email}, \texttt{system\_user.phone\_number}
    \item \texttt{customer.phone\_number}, \texttt{customer.email}
    \item \texttt{drone.serial\_number}
    \item \texttt{landing\_station.station\_code}
    \item \texttt{delivery\_confirmation.order\_id} và \texttt{delivery\_confirmation.activity\_id}
    \item \texttt{delivery\_route.activity\_id}
    \item \texttt{route\_point(route\_id, sequence\_number)}
\end{itemize}

\subsection{Phụ thuộc hàm và chuẩn hóa}

Cả hai mươi ba quan hệ đều thỏa mãn Dạng chuẩn thứ ba (3NF):

\paragraph{Dạng chuẩn thứ nhất (1NF).}
Mọi giá trị cột đều có tính nguyên tử. Các thành phần địa chỉ được tách thành những thuộc tính riêng; kích thước được lưu dưới dạng các giá trị số độc lập (\texttt{length\_cm}, \texttt{width\_cm}, \texttt{height\_cm}); nhiều địa chỉ hoặc kiện hàng được biểu diễn bằng các dòng quan hệ riêng thay vì chuỗi phức hợp.

\paragraph{Dạng chuẩn thứ hai (2NF).}
Mỗi thuộc tính không khóa phụ thuộc hàm đầy đủ vào toàn bộ khóa chính. Trong các bảng có khóa ghép (\texttt{user\_role}, \texttt{role\_permission}), thuộc tính không khóa phụ thuộc vào cả hai thành phần của khóa.

\paragraph{Dạng chuẩn thứ ba (3NF).}
Không tồn tại phụ thuộc bắc cầu giữa các thuộc tính không khóa. Ví dụ, \texttt{delivery\_order} tham chiếu \texttt{customer\_id} và không lưu lại thông tin liên hệ của khách hàng; \texttt{delivery\_confirmation} không lưu dữ liệu khách hàng hoặc máy bay; còn \texttt{tracking\_record} tham chiếu \texttt{activity\_id} mà không lưu dư thừa thuộc tính của máy bay.

\subsection{Các ràng buộc toàn vẹn}

Lược đồ quan hệ thực thi sáu nhóm ràng buộc toàn vẹn:

\begin{enumerate}
    \item \textbf{Toàn vẹn thực thể}: Được bảo đảm bởi các khóa chính thay thế tự động sinh, không rỗng và duy nhất trong từng bảng.
    \item \textbf{Toàn vẹn tham chiếu}: Được thực thi bằng các ràng buộc khóa ngoại. Dữ liệu chủ phục vụ vận hành (\texttt{customer}, \texttt{drone}, \texttt{landing\_station}) sử dụng \texttt{ON DELETE RESTRICT} để ngăn vô tình xóa lịch sử vận hành. Các bảng liên kết bảo mật sử dụng \texttt{ON DELETE CASCADE}. Tham chiếu người dùng trong nhật ký kiểm toán sử dụng \texttt{ON DELETE SET NULL} để bảo toàn dấu vết khi tài khoản được lưu trữ.
    \item \textbf{Toàn vẹn miền}: Kiểu dữ liệu cột và các ràng buộc \texttt{CHECK} tường minh bảo đảm giá trị thuộc miền hợp lệ: tọa độ nằm trong $[-90, 90]$ và $[-180, 180]$, kích thước và tải trọng dương, phần trăm pin từ $0$ đến $100$.
    \item \textbf{Ràng buộc duy nhất}: Các mã định danh tự nhiên và bản số 1:0..1 được thực thi nghiêm ngặt bằng chỉ mục \texttt{UNIQUE} ở cấp cơ sở dữ liệu.
    \item \textbf{Toàn vẹn nghiệp vụ}: Tổng khối lượng kiện hàng không được vượt quá tải trọng máy bay được phân công (BR-18); địa chỉ phải thuộc về khách hàng đặt đơn (BR-04).
    \item \textbf{Toàn vẹn thời gian}: Trong \texttt{delivery\_activity}, \texttt{started\_at < ended\_at}; trong \texttt{station\_operator\_assignment}, \texttt{assigned\_from < assigned\_to}.
\end{enumerate}

\subsection{Mô hình vòng đời chuyển trạng thái}\label{s:lifecycle}

Hệ thống xác định vòng đời chính thức cho ba thực thể trung tâm có trạng thái:

\paragraph{Vòng đời đơn giao hàng.}
\begin{verbatim}
PENDING
   |
   +---> APPROVED ---> ASSIGNED ---> IN_TRANSIT
   |                                     |
   |                                     +---> DELIVERED (Kết thúc)
   |                                     |
   |                                     +---> FAILED
   |                                              |
   |                                              +---> ASSIGNED (Được lập lịch lại)
   |                                              |
   |                                              +---> CANCELLED (Kết thúc)
   |
   +---> REJECTED (Kết thúc)
   |
   +---> CANCELLED (Kết thúc)
\end{verbatim}

\paragraph{Vòng đời kiện hàng.}
\begin{verbatim}
CREATED -> AT_STATION -> PICKED_UP -> IN_TRANSIT -> DELIVERED
                                         |
                                         +-> FAILED -> RETURNED / DAMAGED
\end{verbatim}

\paragraph{Vòng đời hoạt động giao hàng.}
\begin{verbatim}
ASSIGNED -> PREPARING -> IN_TRANSIT -> DELIVERED / FAILED / CANCELLED
\end{verbatim}

\section{Thiết kế cơ sở dữ liệu mức vật lý}\label{s:physical}

Phần này trình bày chi tiết việc triển khai vật lý bằng PostgreSQL. Tất cả các bảng sử dụng \texttt{BIGINT GENERATED ALWAYS AS IDENTITY} cho khóa chính, \texttt{TIMESTAMPTZ} cho trường thời gian và các miền số, hình học phù hợp.

\subsection{Các câu lệnh CREATE TABLE}

\begin{verbatim}
-- 1. Nhóm người dùng và phân quyền

CREATE TABLE app_user (
  user_id       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  username      VARCHAR(100) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  email         VARCHAR(255) UNIQUE,
  phone_number  VARCHAR(20)  UNIQUE,
  is_active     BOOLEAN      NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMPTZ  NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE role (
  role_id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  role_name   VARCHAR(50)  NOT NULL UNIQUE,
  description VARCHAR(255)
);

CREATE TABLE permission (
  permission_id   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  permission_name VARCHAR(100) NOT NULL UNIQUE,
  description     VARCHAR(255)
);

CREATE TABLE user_role (
  user_id     BIGINT      NOT NULL,
  role_id     BIGINT      NOT NULL,
  assigned_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (user_id, role_id),
  CONSTRAINT fk_ur_user FOREIGN KEY (user_id)
    REFERENCES app_user(user_id) ON DELETE CASCADE,
  CONSTRAINT fk_ur_role FOREIGN KEY (role_id)
    REFERENCES role(role_id) ON DELETE CASCADE
);

CREATE TABLE role_permission (
  role_id       BIGINT NOT NULL,
  permission_id BIGINT NOT NULL,
  PRIMARY KEY (role_id, permission_id),
  CONSTRAINT fk_rp_role FOREIGN KEY (role_id)
    REFERENCES role(role_id) ON DELETE CASCADE,
  CONSTRAINT fk_rp_perm FOREIGN KEY (permission_id)
    REFERENCES permission(permission_id) ON DELETE CASCADE
);

-- 2. Nhóm khách hàng và địa chỉ

CREATE TABLE customer (
  customer_id   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id       BIGINT       UNIQUE,
  customer_name VARCHAR(100) NOT NULL,
  phone_number  VARCHAR(20)  UNIQUE,
  email         VARCHAR(255) UNIQUE,
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_customer_user FOREIGN KEY (user_id)
    REFERENCES app_user(user_id) ON DELETE RESTRICT
);

CREATE TABLE customer_address (
  address_id    BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  customer_id   BIGINT       NOT NULL,
  address_label VARCHAR(50),
  address_line  VARCHAR(255) NOT NULL,
  ward          VARCHAR(100),
  district      VARCHAR(100),
  city          VARCHAR(100),
  latitude      NUMERIC(9,6) CHECK (latitude BETWEEN -90 AND 90),
  longitude     NUMERIC(9,6) CHECK (longitude BETWEEN -180 AND 180),
  is_default    BOOLEAN      NOT NULL DEFAULT FALSE,
  created_at    TIMESTAMPTZ  NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_addr_customer FOREIGN KEY (customer_id)
    REFERENCES customer(customer_id) ON DELETE RESTRICT,
  CONSTRAINT uq_customer_address UNIQUE (address_id, customer_id)
);

-- 3. Nhóm trạm hạ cánh và nhân viên vận hành

CREATE TABLE landing_station (
  station_id   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  station_code VARCHAR(50)  NOT NULL UNIQUE,
  station_name VARCHAR(100) NOT NULL,
  address      VARCHAR(255),
  latitude     NUMERIC(9,6) CHECK (latitude BETWEEN -90 AND 90),
  longitude    NUMERIC(9,6) CHECK (longitude BETWEEN -180 AND 180),
  capacity     INTEGER      NOT NULL DEFAULT 0 CHECK (capacity >= 0),

\subsection{Chiến lược chỉ mục}\label{s:indexes}

Chỉ mục được ưu tiên cho khóa ngoại và các điều kiện lọc truy vấn thường xuyên, đồng thời tránh tạo chỉ mục dư thừa trên những cột đã được bao phủ bởi ràng buộc duy nhất:

\begin{verbatim}
-- Chỉ mục khóa ngoại phục vụ các phép nối dữ liệu vận hành
CREATE INDEX idx_order_customer    ON delivery_order(customer_id);

CREATE INDEX idx_osh_order         ON order_status_history(order_id);
CREATE INDEX idx_pkg_order         ON package(order_id);
CREATE INDEX idx_activity_order    ON delivery_activity(order_id);
CREATE INDEX idx_activity_drone    ON delivery_activity(drone_id);
CREATE INDEX idx_activity_prev     ON delivery_activity(previous_activity_id);
CREATE INDEX idx_tracking_activity ON tracking_record(activity_id);
CREATE INDEX idx_exception_activity ON delivery_exception(activity_id);
CREATE INDEX idx_ssh_station       ON station_status_history(station_id);
CREATE INDEX idx_soa_station       ON station_operator_assignment(station_id);
CREATE INDEX idx_soa_user          ON station_operator_assignment(user_id);
CREATE INDEX idx_pse_package       ON package_station_event(package_id);
CREATE INDEX idx_pse_station       ON package_station_event(station_id);
CREATE INDEX idx_audit_user        ON audit_log(user_id);

-- Chỉ mục ghép phục vụ truy vấn dữ liệu đo từ xa gần nhất
CREATE INDEX idx_tracking_activity_time
  ON tracking_record(activity_id, recorded_at DESC);

-- Chỉ mục duy nhất từng phần bảo đảm mỗi khách hàng có tối đa một địa chỉ mặc định (BR-05)
CREATE UNIQUE INDEX uq_customer_default_address
  ON customer_address(customer_id)
  WHERE is_default = TRUE;
\end{verbatim}

\subsection{Các truy vấn nghiệp vụ tiêu biểu}

Các truy vấn sau xác nhận thiết kế vật lý đáp ứng trực tiếp nhu cầu vận hành:

\paragraph{Truy xuất các đơn hàng của khách hàng với trạng thái mới nhất.}
\begin{verbatim}
SELECT order_id, order_status, requested_delivery_time,
       estimated_arrival_at, created_at
FROM   delivery_order
WHERE  customer_id = :customer_id
ORDER  BY created_at DESC;
\end{verbatim}

\paragraph{Truy xuất lịch sử xác nhận lấy kiện hàng tại các trạm.}
\begin{verbatim}
SELECT pse.event_id, pse.package_id, ls.station_name,
       pse.event_type, su.username AS operator_name, pse.occurred_at
FROM   package_station_event pse
JOIN   landing_station ls ON ls.station_id = pse.station_id
LEFT   JOIN app_user su ON su.user_id = pse.performed_by
WHERE  pse.package_id = :package_id
ORDER  BY pse.occurred_at ASC;
\end{verbatim}

\paragraph{Truy vết chuỗi lập lịch lại của một đơn giao hàng.}
\begin{verbatim}
SELECT activity_id, drone_id, activity_status,
       previous_activity_id, started_at, ended_at
FROM   delivery_activity
WHERE  order_id = :order_id
ORDER  BY created_at ASC;
\end{verbatim}

\paragraph{Truy xuất dữ liệu đo từ xa của chuyến bay để giám sát.}
\begin{verbatim}
SELECT recorded_at, latitude, longitude, altitude_m, speed_kmh, battery_level
FROM   tracking_record
WHERE  activity_id = :activity_id
ORDER  BY recorded_at DESC
LIMIT  50;
\end{verbatim}

\section{Tóm tắt chương}\label{s:ch3summary}

\qquad Chương này đã trình bày chi tiết thiết kế cơ sở dữ liệu mức khái niệm, mức logic và mức vật lý cho nền tảng SmartDroneDelivery. Mô hình gồm hai mươi ba quan hệ đã chuẩn hóa, được tổ chức thành bảy nhóm chức năng.

Các cải tiến kiến trúc chính gồm đưa vào \texttt{delivery\_activity} để tách đơn hàng khỏi máy bay; bổ sung \texttt{station\_operator\_assignment} và \texttt{package\_station\_event} để chính thức hóa quy trình bố trí nhân sự tại trạm và lấy kiện hàng; sử dụng \texttt{previous\_activity\_id} để theo dõi chuỗi lập lịch lại; và tham chiếu khóa bằng chứng số bên ngoài trong \texttt{delivery\_confirmation}.

Lược đồ logic thỏa mãn 3NF và thực thi sáu nhóm ràng buộc toàn vẹn. Cuối cùng, chương đã trình bày đầy đủ các câu lệnh DDL PostgreSQL và chiến lược chỉ mục, cung cấp một triển khai chặt chẽ, hiệu quả, có khả năng truy vết đầy đủ và sẵn sàng được kiểm chứng toàn diện trong Chương~\ref{c:test}.
